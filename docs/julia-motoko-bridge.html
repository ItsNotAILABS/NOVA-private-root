JuliaCompute.mo — Door 2: Motoko Smart Contract 
// Door 2: Motoko / Candid — Smart Contract Interface
import JuliaCompute "JuliaCompute";
import Float "mo:base/Float";

actor MyCanister {
    // Typed call to Julia bridge
    public shared func computeSpectrum(
        matrix: [[Float]]
    ) : async JuliaCompute.EigenResult {
        await JuliaCompute.linalg_eigen(matrix)
    };

    // Kuramoto coherence check
    public shared func checkCoherence(
        oscillators: [JuliaCompute.Oscillator]
    ) : async Bool {
        let R = await JuliaCompute.order_parameter(oscillators);
        R > JuliaCompute.PHI_INV  // φ⁻¹ threshold
    };
}

NovaJulia.jl — Julia Mathematical Substrate 
# NOVA Julia Mathematical Substrate
using LinearAlgebra, Statistics, FFTW

# Golden ratio constant
const PHI = 1.618033988749894
const PHI_INV = 0.6180339887498949

# φ-optimized eigenvalue computation
function phi_eigen(A::Matrix{Float64})
    vals, vecs = eigen(A)
    return (
        values = vals .* PHI_INV,  # φ-scaled
        vectors = vecs
    )
end

# Kuramoto φ-oscillator synchronization
function kuramoto_sync(θ::Vector, K::Float64, ω::Vector)
    N = length(θ)
    R = abs(mean(exp.(im .* θ)))
    dθ = ω .+ (K * R / N) .* sin.(mean(θ) .- θ)
    return dθ .* PHI_INV  # φ-damped
end
